Differentiate each function (with respect to \(x\))
(a)
\(\cos \left ( \sqrt {x+7} \right ) \)
\begin{align*} \dfrac {d}{dx} \left ( \cos \left ( \sqrt {x+7} \right ) \right ) &= -\sin \left ( \sqrt {x+7} \right ) \cdot \frac {1}{2} (x+7)^{\frac {-1}{2}} (1) \\ &= \frac {-\sin \left ( \sqrt {x+7} \right ) }{2 \sqrt {x+7}} \end{align*}
(b)
\(\sqrt { \cos (x) + 7}\)
\begin{align*} \dfrac {d}{dx} \left ( \sqrt { \cos (x) + 7} \right ) &= \frac {1}{2} \left ( \cos (x) + 7 \right )^{\frac {-1}{2}} \left ( -\sin (x) \right ) \\ &= \frac {- \sin (x) }{2 \sqrt {\cos (x) + 7 }} \end{align*}
(c)
\(\sqrt {\cos (x)} + 7\)
\begin{align*} \dfrac {d}{dx} \left ( \sqrt {\cos (x)} + 7 \right ) &= \frac {1}{2} \left ( \cos (x) \right )^{\frac {-1}{2}} \left ( -\sin (x) \right ) + 0 \\ &= \frac {- \sin (x)}{2 \sqrt {\cos (x)}} \end{align*}
(d)
\(\cos \left ( \sqrt {x} + 7 \right )\)
\begin{align*} \dfrac {d}{dx} \left ( \cos \left ( \sqrt {x} + 7 \right ) \right ) &= - \sin \left ( \sqrt {x} + 7 \right ) \cdot \frac {1}{2} x^{\frac {-1}{2}} \\ &= \frac {- \sin \left ( \sqrt {x} + 7 \right ) }{2 \sqrt {x}} \end{align*}
(e)
\(\cos (x) \cdot \left ( \sqrt {x} + 7 \right ) \)
\begin{align*} \dfrac {d}{dx}\left (\cos (x)\cdot \left ( \sqrt {x} + 7 \right ) \right ) &= - \sin (x)\cdot \left ( \sqrt {x} + 7 \right )+\cos (x)\cdot \frac {1}{2} x^{\frac {-1}{2}} \\ &= - \sin (x)\cdot \left ( \sqrt {x} + 7 \right )+\frac {\cos (x)}{2\sqrt {x}} \end{align*}
