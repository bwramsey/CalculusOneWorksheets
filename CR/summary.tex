\begin{itemize}
	\item \textbf{The Chain Rule}: If $f(x)$ and $g(x)$ are differentiable functions, then $\displaystyle \dfrac{d}{dx} f\left(g(x)\right) = f'\left( g(x)\right) g'(x)$.
		
	\item \textbf{The Derivatives of Trigonometric Functions}:
		\begin{align*}
			&\dfrac{d}{dx}\sin(x) = \cos(x)  \quad & \dfrac{d}{dx} \cos(x) = -\sin(x) \\
			&\dfrac{d}{dx}\tan(x) = \sec^2(x)  &\dfrac{d}{dx} \cot(x) = -\csc^2(x) \\
			&\dfrac{d}{dx}\sec(x) = \sec(x)\tan(x)  &\dfrac{d}{dx} \csc(x) = -\csc(x)\cot(x)
		\end{align*}
\end{itemize}
