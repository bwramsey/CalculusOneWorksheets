\begin{problem}

State the form of the limit. Determine whether the form is determinate or indeterminate. Evaluate each limit.  
  \begin{enumerate}
    \item  $\displaystyle \lim_{x \to \infty}\left( \ln (1 + e^{-x}) \right)^x $\\
    \item  $\displaystyle \lim_{x \to \infty} \left( \dfrac{1}{x} + 1 \right)^{\frac{1}{x}} $\\
    \item  $\displaystyle \lim_{x \to \infty} \left( \dfrac{\arctan(x)}{x} \right) $\\
    \item  $\displaystyle \lim_{x \to \infty} (x-\ln(x)) $\\
    \item  $\displaystyle \lim_{x \to \infty} \left( x \ln \left( \dfrac{1}{x} \right) \right) $\\
    \item  $\displaystyle\lim_{x\to0^+} (\sin(x) \cot(x) ) $
    \end{enumerate}
\end{problem}
