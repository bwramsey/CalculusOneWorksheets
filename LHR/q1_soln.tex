\begin{problem}
State the form of the limit. Determine whether the form is determinate or indeterminate. Evaluate each limit.  
  \begin{enumerate}
      \item  $\displaystyle \lim_{x \to 0}\dfrac{\sin(x)-\cos(x)+1}{x^2-x}$
    \begin{explanation}
    The form is $\dfrac{0}{0}$; indeterminate. we can apply L.R.\\
     $\displaystyle \lim_{x \to 0}\dfrac{\sin(x)-\cos(x)+1}{x^2-x}=\displaystyle \lim_{x \to 0}\dfrac{\cos(x)+\sin(x)}{2x-1}=\dfrac{\cos(0)+\sin(0)}{2(0)-1}=\dfrac{1}{-1}=-1$

    \end{explanation}
  \item  $\displaystyle \lim_{x \to 0}\dfrac{e^x-1-x}{x^2}$
    \begin{explanation}
    The form is $\dfrac{0}{0}$; indeterminate.\\
     $\displaystyle \lim_{x \to 0}\dfrac{e^x-1-x}{x^2}= \displaystyle \lim_{x \to 0}\dfrac{e^x-1}{2x}$, by L.R.\\
     The last limit also has  the form $\dfrac{0}{0}$, so, we can apply L.R. again.\\
     $\displaystyle \lim_{x \to 0}\dfrac{e^x-1-x}{x^2}= \displaystyle \lim_{x \to 0}\dfrac{e^x-1}{2x}= \displaystyle \lim_{x \to 0}\dfrac{e^x}{2}=\dfrac{1}{2}$
    \end{explanation}
     \item  $\displaystyle \lim_{x \to \infty}\dfrac{e^x}{x^4}$
    \begin{explanation}
    The form is $\dfrac{\infty}{\infty}$; indeterminate.\\
    $\displaystyle \lim_{x \to \infty}\dfrac{e^x}{x^4}=\displaystyle \lim_{x \to \infty}\dfrac{e^x}{4x^3}$ , by L.R.\\
     The last limit also has  the form $\dfrac{\infty}{\infty}$. It turns out that we can apply L.R. again, and again, and again:\\
     $\displaystyle \lim_{x \to \infty}\dfrac{e^x}{x^4}=\displaystyle \lim_{x \to \infty}\dfrac{e^x}{4x^3}= \displaystyle \lim_{x \to \infty}\dfrac{e^x}{12x^2}= \displaystyle \lim_{x \to \infty}\dfrac{e^x}{24x}= \displaystyle \lim_{x \to \infty}\dfrac{e^x}{24}=\infty$\\
     This result shows that the function $e^x$ grows much faster than $x^4$, as $x$ goes to $\infty$.
        \end{explanation}
  
      \end{enumerate}
      \end{problem}
